```python
import streamlit as st
import pandas as pd
import requests
from streamlit_gsheets import GSheetsConnection
from datetime import datetime

# --- KONFIGURASI TEMA ---
st.set_page_config(page_title="PREDIKSI PARLAY AI", page_icon="⚽", layout="wide")

# Koneksi ke Google Sheets (Database)
# Note: Ensure 'gsheets' connection is configured in Streamlit Secrets for deployment
try:
    conn = st.connection("gsheets", type=GSheetsConnection)
except Exception:
    conn = None

st.title("🤖 PREDIKSI PARLAY AI")
st.markdown("---")

# --- SIDEBAR PENGATURAN ---
st.sidebar.header("⚙️ PENGATURAN")
api_key = st.sidebar.text_input("The Odds API Key", type="password", help="Get your free key from the-odds-api.com")
min_odds_val = st.sidebar.number_input("Minimal Odds per Leg", value=1.80, step=0.10)
league = st.sidebar.selectbox("Pilih Liga", ["soccer_epl", "soccer_spain_la_liga", "soccer_italy_serie_a", "soccer_germany_bundesliga"])

if 'parlay_list' not in st.session_state:
    st.session_state.parlay_list = []

if 'current_matches' not in st.session_state:
    st.session_state.current_matches = []

# --- FUNGSI AMBIL DATA ---
def fetch_odds(key, league_code):
    url = f"https://api.the-odds-api.com/v4/sports/{league_code}/odds/?apiKey={key}&regions=eu&markets=h2h"
    try:
        res = requests.get(url)
        return res.json() if res.status_code == 200 else None
    except Exception as e:
        st.error(f"Error fetching data: {e}")
        return None

# --- KOLOM UTAMA ---
col_data, col_parlay = st.columns([2, 1])

with col_data:
    st.subheader("📋 Analisis Pertandingan")
    
    if st.button("🔄 Perbarui Data Pertandingan"):
        if api_key:
            data = fetch_odds(api_key, league)
            if data:
                matches = []
                for m in data:
                    home = m['home_team']
                    away = m['away_team']
                    # Check if bookmakers and markets exist to avoid errors
                    if m['bookmakers'] and m['bookmakers'][0]['markets']:
                        for outcome in m['bookmakers'][0]['markets'][0]['outcomes']:
                            if outcome['price'] >= min_odds_val:
                                matches.append({
                                    "Pertandingan": f"{home} vs {away}",
                                    "Pilihan": outcome['name'],
                                    "Odds": outcome['price'],
                                    "Waktu": m['commence_time']
                                })
                st.session_state.current_matches = matches
                st.success(f"Ditemukan {len(matches)} pilihan potensial.")
            else:
                st.error("Gagal mengambil data. Periksa API Key atau kuota API Anda.")
        else:
            st.warning("Masukkan API Key di sidebar terlebih dahulu.")

    if st.session_state.current_matches:
        df = pd.DataFrame(st.session_state.current_matches)
        
        # Pilihan Interaktif
        if not df.empty:
            option_list = df['Pertandingan'] + " (" + df['Pilihan'] + ") - Odds: " + df['Odds'].astype(str)
            selected_match_str = st.selectbox("Pilih pertandingan untuk masuk Parlay:", option_list)
            
            if st.button("➕ Tambahkan ke Parlay"):
                # Extract original data based on selection
                selected_idx = option_list[option_list == selected_match_str].index[0]
                selected_item = df.iloc[selected_idx].to_dict()
                st.session_state.parlay_list.append(selected_item)
                st.success(f"Berhasil menambahkan {selected_item['Pertandingan']}")
        
        st.dataframe(df, use_container_width=True)
    else:
        st.info("Klik 'Perbarui Data Pertandingan' untuk memuat daftar odds terbaru.")

with col_parlay:
    st.subheader("💰 Tiket Parlay Anda")
    if st.session_state.parlay_list:
        total_odds = 1.0
        details = []
        for item in st.session_state.parlay_list:
            st.write(f"✅ {item['Pertandingan']} ({item['Odds']})")
            total_odds *= item['Odds']
            details.append(f"{item['Pertandingan']}@{item['Odds']}")
        
        st.markdown("---")
        st.metric("Total Odds", f"{total_odds:.2f}x")
        
        stake = st.number_input("Modal Taruhan (Rp/USD)", min_value=0, value=100000)
        potential_win = stake * total_odds
        st.write(f"Estimasi Kemenangan: **Rp {potential_win:,.2f}**")
        
        col_save, col_reset = st.columns(2)
        with col_save:
            if st.button("💾 SIMPAN KE RIWAYAT"):
                if conn:
                    try:
                        # Read existing data
                        try:
                            existing_df = conn.read(worksheet="Sheet1", ttl=5)
                        except Exception:
                            existing_df = pd.DataFrame(columns=["Tanggal", "Detail Parlay", "Total Odds", "Modal", "Estimasi", "Status"])
                        
                        # Create new entry
                        new_entry = pd.DataFrame([{
                            "Tanggal": datetime.now().strftime("%Y-%m-%d %H:%M"),
                            "Detail Parlay": " | ".join(details),
                            "Total Odds": round(total_odds, 2),
                            "Modal": stake,
                            "Estimasi": round(potential_win, 2),
                            "Status": "Pending"
                        }])
                        
                        # Append and save
                        updated_df = pd.concat([existing_df, new_entry], ignore_index=True)
                        conn.update(worksheet="Sheet1", data=updated_df)
                        st.success("Taruhan Berhasil Disimpan ke Google Sheets!")
                        st.session_state.parlay_list = [] # Reset after save
                        st.rerun()
                    except Exception as e:
                        st.error(f"Gagal menyimpan ke Sheets: {e}. Pastikan konfigurasi Secrets benar.")
                else:
                    st.warning("Koneksi Database belum dikonfigurasi. Simpan manual atau setup Secrets.")
        
        with col_reset:
            if st.button("🗑️ Reset"):
                st.session_state.parlay_list = []
                st.rerun()
                
    else:
        st.write("Belum ada tim yang dipilih.")

# --- BAGIAN DATABASE (RIWAYAT) ---
st.markdown("---")
st.subheader("📊 Riwayat Taruhan Anda")

if conn:
    try:
        existing_data = conn.read(worksheet="Sheet1", ttl=5)
        if existing_data is not None and not existing_data.empty:
            st.dataframe(existing_data.sort_values(by="Tanggal", ascending=False), use_container_width=True)
        else:
            st.info("Belum ada riwayat taruhan tersimpan di database.")
    except Exception as e:
        st.error(f"Gagal memuat riwayat: {e}")
else:
    st.info("Mode Lokal: Riwayat tidak ditampilkan karena koneksi Google Sheets belum aktif.")

st.markdown("---")
st.caption("© 2026 PREDIKSI PARLAY AI - Developed for Randy Sanjaya")
```


DEPLOYMENT INSTRUCTIONS
=======================

1. LOCAL SETUP
   - Install Python if not already installed.
   - Open terminal/command prompt.
   - Navigate to the folder containing app.py and requirements.txt.
   - Run: pip install -r requirements.txt
   - Run: streamlit run app.py
   - Access the app at http://localhost:8501

2. ONLINE DEPLOYMENT (STREAMLIT CLOUD)
   - Create a GitHub repository and upload app.py and requirements.txt.
   - Go to share.streamlit.io and login with GitHub.
   - Click "New App" and select your repository.
   - Set Main file path to app.py.
   - Click Deploy.

3. GOOGLE SHEETS CONFIGURATION (FOR ONLINE APP)
   - Create a new Google Sheet with headers: Tanggal, Detail Parlay, Total Odds, Modal, Estimasi, Status.
   - In Streamlit Cloud, go to Settings > Secrets.
   - Add the following secret (replace with your actual sheet URL):
     
     [connections.gsheets]
     spreadsheet = "YOUR_GOOGLE_SHEET_URL_HERE"

   - Save secrets. The app will now save betting history to your sheet.

{match_info['Pertandingan']}")

with col_parlay:
    st.subheader("💰 Tiket Parlay Anda")
    
    if st.session_state.parlay_list:
        total_odds = 1.0
        details = []
        
        st.write("**Daftar Pilihan:**")
        for item in st.session_state.parlay_list:
            st.write(f"✅ {item['Pertandingan']} ({item['Odds']})")
            total_odds *= item['Odds']
            details.append(f"{item['Pertandingan']}@{item['Odds']}")
        
        st.markdown("---")
        st.metric("Total Odds", f"{total_odds:.2f}x")
        
        stake = st.number_input("Modal Taruhan (Rp/$)", value=100000, step=10000)
        potential_win = stake * total_odds
        st.write(f"Estimasi Kemenangan: **Rp {potential_win:,.2f}**")
        
        if st.button("💾 SIMPAN KE RIWAYAT"):
            # Initialize connection
            try:
                conn = st.connection("gsheets", type=GSheetsConnection)
                success = save_to_history(conn, details, total_odds, stake)
                
                if success:
                    st.success("Taruhan berhasil disimpan ke Database!")
                    st.session_state.parlay_list = [] # Clear list after saving
                    st.rerun()
                else:
                    st.error("Gagal menyimpan. Periksa konfigurasi Google Sheets.")
            except Exception as e:
                st.error(f"Koneksi Database Error: {e}")
                st.info("Pastikan Anda telah mengatur 'Secrets' di Streamlit Cloud dengan URL Google Sheets Anda.")
                
        if st.button("🗑️ Reset Parlay"):
            st.session_state.parlay_list = []
            st.rerun()
            
    else:
        st.info("Belum ada tim yang dipilih. Silakan pilih dari daftar pertandingan.")
